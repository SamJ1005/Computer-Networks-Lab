import socket
import threading
import sys

HOST = '0.0.0.0'
PORT = 65432

# We use this to keep track of what the user is currently typing
current_input = ""

def receive_messages(conn):
    while True:
        try:
            data = conn.recv(1024).decode('utf-8')
            if not data: break
            
            # 1. Clear the line you are typing on
            # 2. Print the incoming message
            # 3. Put back "You (Device 1): " AND whatever characters you already typed
            sys.stdout.write(f"\r\033[KDevice 2: {data}\n")
            sys.stdout.write(f"You (Device 1): {current_input}")
            sys.stdout.flush()
        except:
            break

def start_server():
    global current_input
    server = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    server.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
    server.bind((HOST, PORT))
    server.listen(1)
    print(f"Server waiting on {PORT}...")

    conn, addr = server.accept()
    print(f"Connected to {addr}")

    threading.Thread(target=receive_messages, args=(conn,), daemon=True).start()

    while True:
        # Instead of a simple input(), we build the string or use the prompt
        msg = input(f"You (Device 1): ")
        # While typing, current_input isn't updated in real-time with standard input()
        # but once you hit Enter, we send it.
        conn.sendall(msg.encode('utf-8'))
        current_input = "" # Reset after sending

if __name__ == "__main__":
    start_server()

import socket
import threading
import sys

SERVER_IP = '127.0.0.1' # Change to Device 1's IP
PORT = 65432
current_input = ""

def receive_messages(client_socket):
    while True:
        try:
            data = client_socket.recv(1024).decode('utf-8')
            if not data: break
            
            # This logic ensures the incoming message "pushes" your typing down 
            # without deleting it.
            sys.stdout.write(f"\r\033[KDevice 1: {data}\n")
            sys.stdout.write(f"You (Device 2): {current_input}")
            sys.stdout.flush()
        except:
            break

def start_client():
    global current_input
    client = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    try:
        client.connect((SERVER_IP, PORT))
        print(f"Connected to {SERVER_IP}")

        threading.Thread(target=receive_messages, args=(client,), daemon=True).start()

        while True:
            msg = input(f"You (Device 2): ")
            client.sendall(msg.encode('utf-8'))
            current_input = "" 
    except Exception as e:
        print(f"Error: {e}")
    finally:
        client.close()

if __name__ == "__main__":
    start_client()
