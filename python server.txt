import socket
import threading
import sys
import termios
import tty
import select

HOST = '0.0.0.0'
PORT = 65432

def get_char():
    """Non-blocking single character input"""
    fd = sys.stdin.fileno()
    old_settings = termios.tcgetattr(fd)
    try:
        tty.setraw(fd)
        if select.select([sys.stdin], [], [], 0.1)[0]:
            return sys.stdin.read(1)
    finally:
        termios.tcsetattr(fd, termios.TCSADRAIN, old_settings)
    return None

def receive_messages(conn):
    while True:
        try:
            data = conn.recv(1024).decode('utf-8')
            if not data: break
            print(f"\r\033[KDevice 2: {data}\nYou (Device 1): ", end='', flush=True)
        except:
            break

def start_server():
    server = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    server.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
    server.bind((HOST, PORT))
    server.listen(1)
    print(f"Server waiting on {PORT}...")
    
    conn, addr = server.accept()
    print(f"Connected to {addr}\n")
    
    threading.Thread(target=receive_messages, args=(conn,), daemon=True).start()
    
    current_input = ""
    while True:
        char = get_char()
        if char == '\r' or char == '\n':  # Enter pressed
            print(f"\nYou (Device 1): {current_input}")
            conn.sendall(current_input.encode('utf-8'))
            current_input = ""
        elif char == '\x7f':  # Backspace
            if current_input:
                current_input = current_input[:-1]
                sys.stdout.write('\b \b')
                sys.stdout.flush()
        elif char:
            current_input += char
            sys.stdout.write(char)
            sys.stdout.flush()

if __name__ == "__main__":
    start_server()

import socket
import threading
import sys
import termios
import tty
import select

SERVER_IP = '127.0.0.1'  # Change to Device 1's IP
PORT = 65432

def get_char():
    """Non-blocking single character input"""
    fd = sys.stdin.fileno()
    old_settings = termios.tcgetattr(fd)
    try:
        tty.setraw(fd)
        if select.select([sys.stdin], [], [], 0.1)[0]:
            return sys.stdin.read(1)
    finally:
        termios.tcsetattr(fd, termios.TCSADRAIN, old_settings)
    return None

def receive_messages(client_socket):
    while True:
        try:
            data = client_socket.recv(1024).decode('utf-8')
            if not data: 
                break
            # Clear line, show message, redraw prompt
            sys.stdout.write(f"\r\033[KDevice 1: {data}\nYou (Device 2): ")
            sys.stdout.flush()
        except:
            break

def start_client():
    client = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    try:
        client.connect((SERVER_IP, PORT))
        print(f"Connected to {SERVER_IP}\n")

        # Start receiving messages in background thread
        threading.Thread(target=receive_messages, args=(client,), daemon=True).start()

        # Main input loop
        current_input = ""
        print("You (Device 2): ", end='', flush=True)
        
        while True:
            char = get_char()
            if char == '\r' or char == '\n':  # Enter pressed
                if current_input.strip():  # Only send non-empty messages
                    print(f"\n{current_input}")
                    client.sendall(current_input.encode('utf-8'))
                    current_input = ""
                print("You (Device 2): ", end='', flush=True)
            elif char == '\x7f':  # Backspace
                if current_input:
                    current_input = current_input[:-1]
                    sys.stdout.write('\b \b')
                    sys.stdout.flush()
            elif char:
                current_input += char
                sys.stdout.write(char)
                sys.stdout.flush()
                
    except Exception as e:
        print(f"Error: {e}")
    finally:
        client.close()

if __name__ == "__main__":
    start_client()
