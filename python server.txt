import socket
import threading
import sys
import msvcrt
import time

HOST = '0.0.0.0'
PORT = 65432

def get_char_windows():
    """Windows-compatible non-blocking input"""
    if msvcrt.kbhit():
        return msvcrt.getch().decode('utf-8', errors='ignore')
    return None

def receive_messages(conn):
    while True:
        try:
            conn.settimeout(0.1)
            data = conn.recv(1024).decode('utf-8')
            if not data: break
            sys.stdout.write(f"\r\033[KDevice 2: {data}\nYou (Device 1): ")
            sys.stdout.flush()
        except socket.timeout:
            continue
        except:
            break

def start_server():
    server = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    server.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
    server.bind((HOST, PORT))
    server.listen(1)
    print(f"Server waiting on {PORT}...")
    
    conn, addr = server.accept()
    print(f"Connected to {addr}\n")
    
    threading.Thread(target=receive_messages, args=(conn,), daemon=True).start()
    
    current_input = ""
    print("You (Device 1): ", end='', flush=True)
    
    while True:
        char = get_char_windows()
        if char == '\r':  # Enter on Windows
            if current_input.strip():
                print(f"\n{current_input}")
                conn.sendall(current_input.encode('utf-8'))
            current_input = ""
            print("You (Device 1): ", end='', flush=True)
        elif char == '\x08':  # Backspace
            if current_input:
                current_input = current_input[:-1]
                sys.stdout.write('\b \b')
                sys.stdout.flush()
        elif char:
            current_input += char
            sys.stdout.write(char)
            sys.stdout.flush()
        time.sleep(0.01)  # Prevent CPU hogging

if __name__ == "__main__":
    start_server()


import socket
import threading
import sys
import msvcrt
import time

SERVER_IP = '127.0.0.1'  # Change to actual server IP
PORT = 65432

def get_char_windows():
    """Windows-compatible non-blocking input"""
    if msvcrt.kbhit():
        return msvcrt.getch().decode('utf-8', errors='ignore')
    return None

def receive_messages(client_socket):
    while True:
        try:
            client_socket.settimeout(0.1)
            data = client_socket.recv(1024).decode('utf-8')
            if not data: break
            sys.stdout.write(f"\r\033[KDevice 1: {data}\nYou (Device 2): ")
            sys.stdout.flush()
        except socket.timeout:
            continue
        except:
            break

def start_client():
    client = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    try:
        client.connect((SERVER_IP, PORT))
        print(f"Connected to {SERVER_IP}\n")

        threading.Thread(target=receive_messages, args=(client,), daemon=True).start()

        current_input = ""
        print("You (Device 2): ", end='', flush=True)
        
        while True:
            char = get_char_windows()
            if char == '\r':  # Enter on Windows
                if current_input.strip():
                    print(f"\n{current_input}")
                    client.sendall(current_input.encode('utf-8'))
                current_input = ""
                print("You (Device 2): ", end='', flush=True)
            elif char == '\x08':  # Backspace
                if current_input:
                    current_input = current_input[:-1]
                    sys.stdout.write('\b \b')
                    sys.stdout.flush()
            elif char:
                current_input += char
                sys.stdout.write(char)
                sys.stdout.flush()
            time.sleep(0.01)
                
    except Exception as e:
        print(f"Error: {e}")
    finally:
        client.close()

if __name__ == "__main__":
    start_client()
