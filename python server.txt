import socket
import threading
import sys

HOST = '0.0.0.0'
PORT = 65432
clients = []

def broadcast(message, sender):
    """Send message to all clients except sender"""
    for client in clients[:]:
        try:
            client.send(message.encode('utf-8'))
        except:
            clients.remove(client)
            client.close()

def handle_client(client, addr):
    print(f"\n✅ Device 2 ({addr}) joined chat!")
    clients.append(client)
    
    while True:
        try:
            message = client.recv(1024).decode('utf-8')
            if not message: break
            print(f"\n👤 Device 2: {message}")
        except:
            break
    
    clients.remove(client)
    client.close()
    print("\n📴 Device 2 left chat")

def send_messages(client):
    """Device 1 sends messages"""
    while True:
        try:
            message = input("\n💬 You (Device 1): ")
            if message.lower() == 'quit':
                break
            broadcast(f"Device 1: {message}", client)
        except:
            break

# Start server
server = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
server.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
server.bind((HOST, PORT))
server.listen(1)

print("🚀 Server ready on port 65432...")
print("⏳ Waiting for Device 2...")

client, addr = server.accept()
print("\n🎉 Connection established!")

# Start client handler & Device 1 input
threading.Thread(target=handle_client, args=(client, addr), daemon=True).start()
threading.Thread(target=send_messages, args=(client,), daemon=True).start()

try:
    while True:
        pass
except KeyboardInterrupt:
    print("\n👋 Server shutdown")
    server.close()


import socket
import threading

SERVER_IP = '127.0.0.1'  # Change to server IP
PORT = 65432

def receive_messages(client):
    print("\n✅ Connected to chat!")
    while True:
        try:
            message = client.recv(1024).decode('utf-8')
            if not message: break
            print(f"\n👤 {message}")
        except:
            break

def send_messages(client):
    while True:
        try:
            message = input("\n💬 You (Device 2): ")
            if message.lower() == 'quit':
                break
            client.send(message.encode('utf-8'))
        except:
            break

# Connect
client = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
try:
    client.connect((SERVER_IP, PORT))
    
    # Separate threads for receive/send - NO CONFLICTS
    threading.Thread(target=receive_messages, args=(client,), daemon=True).start()
    threading.Thread(target=send_messages, args=(client,), daemon=True).start()
    
    while True:
        pass
except KeyboardInterrupt:
    print("\n👋 Disconnected")
finally:
    client.close()
