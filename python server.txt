import socket
import threading

HOST = '0.0.0.0'
PORT = 65432

def handle_client(client_socket, addr):
    print(f"\n📱 Device 2 ({addr}) connected!")
    
    while True:
        try:
            message = client_socket.recv(1024).decode('utf-8')
            if not message:
                break
            print(f"👤 Device 2: {message}")
        except:
            break
    
    client_socket.close()
    print(f"📱 Device 2 ({addr}) disconnected")

def receive_input(client_socket):
    while True:
        try:
            message = input("💬 You (Device 1): ")
            if message.lower() == 'quit':
                break
            client_socket.send(message.encode('utf-8'))
        except:
            break

server = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
server.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
server.bind((HOST, PORT))
server.listen()

print("🚀 Server started on port 65432")
print("⏳ Waiting for Device 2...")

client_socket, addr = server.accept()
print("\n✅ Connected!")

# Start threads
threading.Thread(target=handle_client, args=(client_socket, addr), daemon=True).start()
threading.Thread(target=receive_input, args=(client_socket,), daemon=True).start()

try:
    while True:
        pass
except KeyboardInterrupt:
    print("\n👋 Shutting down...")


import socket
import threading

SERVER_IP = '127.0.0.1'  # Change to server IP
PORT = 65432

def receive_messages(client_socket):
    print("\n✅ Connected to server!")
    while True:
        try:
            message = client_socket.recv(1024).decode('utf-8')
            if not message:
                break
            print(f"👤 Device 1: {message}")
        except:
            break

def send_messages(client_socket):
    while True:
        try:
            message = input("💬 You (Device 2): ")
            if message.lower() == 'quit':
                break
            client_socket.send(message.encode('utf-8'))
        except:
            break

# Connect
client = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
client.connect((SERVER_IP, PORT))

# Start threads
threading.Thread(target=receive_messages, args=(client,), daemon=True).start()
threading.Thread(target=send_messages, args=(client,), daemon=True).start()

try:
    while True:
        pass
except KeyboardInterrupt:
    print("\n👋 Disconnected")
