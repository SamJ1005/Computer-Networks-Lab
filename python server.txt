import socket
import threading
import sys

KEY = "secretkey"

def encrypt_decrypt(message, key):
    encrypted = ""
    for i in range(len(message)):
        encrypted += chr(ord(message[i]) ^ ord(key[i % len(key)]))
    return encrypted

def receive_messages(conn):
    while True:
        try:
            data = conn.recv(1024).decode()
            if not data:
                break
            decrypted = encrypt_decrypt(data, KEY)

            print("\n\nClient:", decrypted)
            print("You: ", end="", flush=True)

        except:
            break

def send_messages(conn):
    while True:
        message = input("You: ")
        encrypted = encrypt_decrypt(message, KEY)
        conn.send(encrypted.encode())

server = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
server.bind(("0.0.0.0", 5000))
server.listen(1)

print("Server started. Waiting for connection...")
conn, addr = server.accept()
print("Connected to:", addr)
print("You can start chatting.\n")

threading.Thread(target=receive_messages, args=(conn,), daemon=True).start()
send_messages(conn)
